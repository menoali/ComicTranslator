#!/usr/bin/env python3 “”” AutoScanlate AR — الاستخدام: python3 manga_translate.py input.jpg [output.jpg] يحتاج: pip install opencv-python-headless pillow “”” import cv2, numpy as np, sys, os from PIL import Image, ImageDraw, ImageFont FONT_PATH = ‘/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf’ def gf(sz): try: return ImageFont.truetype(FONT_PATH, max(8,int(sz))) except: return ImageFont.load_default() def wt(draw, txt, font, mw): words=txt.split(); cx,cy=(x1+x2)/2,(y1+y2)/2 lum=bg[0]*.299+bg[1]*.587+bg[2]*.114 tc=(10,8,5) if lum>100 else (240,235,215) pil=Image.fromarray(cv2.cvtColor(img_cv,cv2.COLOR_BGR2RGB)) draw=ImageDraw.Draw(pil) best_fs,best_lines,best_lh=9,[txt],12 for fs in range(int(bh*.52),8,-1): font=gf(fs);
